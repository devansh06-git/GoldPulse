import numpy as np
import pandas as pd
from typing import Dict, List, Optional
from fastapi import FastAPI, UploadFile, File, HTTPException

app = FastAPI(title="MCX Gold RV Engine", version="2.4.0")

CONTRACT_SPECS = {
    'GOLDM':      {'lot_g': 100, 'quote_g': 10, 'purity': 995},
    'GOLDTEN':    {'lot_g': 10,  'quote_g': 10, 'purity': 999},
    'GOLDGUINEA': {'lot_g': 8,   'quote_g': 8,  'purity': 999},
    'GOLDPETAL':  {'lot_g': 1,   'quote_g': 1,  'purity': 999}
}

COST_THRESHOLD_PCT = 0.0010  # 0.10% (10 bps) brokerage, STT & slippage
MIN_VOLUME_THRESHOLD = 100   # Filter out illiquid stale quotes

class MCXQuantEngine:
    @staticmethod
    def normalize_row(row: pd.Series) -> float:
        symbol = str(row['SYMBOL']).strip().upper()
        close_price = float(row['CLOSE'])
        
        if symbol not in CONTRACT_SPECS:
            return np.nan
            
        spec = CONTRACT_SPECS[symbol]
        base_1g = close_price / spec['quote_g']
        purity_adj_1g = base_1g * (999.0 / spec['purity'])
        return round(purity_adj_1g, 4)

    @classmethod
    def process_bhavcopy(cls, df: pd.DataFrame) -> pd.DataFrame:
        df = df.copy()
        df['SYMBOL'] = df['SYMBOL'].astype(str).str.strip().str.upper()
        df = df[df['SYMBOL'].isin(CONTRACT_SPECS.keys())]
        
        # 1. Normalize Prices
        df['NORM_1G_PRICE'] = df.apply(cls.normalize_row, axis=1)
        
        # 2. Term-Structure Liquidity Weighted Benchmark Price
        liquid_mask = df['VOLUME'] >= MIN_VOLUME_THRESHOLD
        df_liquid = df[liquid_mask]
        
        benchmarks = df_liquid.groupby('EXPIRY_DT')['NORM_1G_PRICE'].mean().to_dict()
        df['BENCHMARK_1G'] = df['EXPIRY_DT'].map(benchmarks)
        
        # 3. Calculate Relative Deviation & Signals
        df['DEV_PCT'] = ((df['NORM_1G_PRICE'] - df['BENCHMARK_1G']) / df['BENCHMARK_1G'])
        
        def assign_signal(row):
            if row['VOLUME'] < MIN_VOLUME_THRESHOLD:
                return 'THIN LIQUIDITY'
            if pd.isna(row['DEV_PCT']):
                return 'QUIET / Neutral'
            if row['DEV_PCT'] > COST_THRESHOLD_PCT:
                return 'OVERPRICED'
            elif row['DEV_PCT'] < -COST_THRESHOLD_PCT:
                return 'UNDERPRICED'
            return 'QUIET / Neutral'
            
        df['SIGNAL'] = df.apply(assign_signal, axis=1)
        return df

@app.post("/api/v1/normalize")
async def process_mcx_file(file: UploadFile = File(...)):
    if not file.filename.endswith('.csv'):
        raise HTTPException(status_code=400, detail="Invalid file type. CSV required.")
    
    df_raw = pd.read_csv(file.file)
    df_processed = MCXQuantEngine.process_bhavcopy(df_raw)
    return df_processed.to_dict(orient='records')
